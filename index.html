Niche: "interesting facts". Topic: "{topic}" Requirements: - Hook in FIRST 3 seconds - ~{target_words} words total (55 seconds spoken) - Fast punchy sentences, no fluff - End with call to action Respond ONLY with valid JSON (no markdown): {{ "title": "catchy title max 100 chars", "hook": "opening 1-2 sentences", "script": "full spoken script ~{target_words} words", "description": "2-3 sentence YouTube description", "hashtags": "#fact1 #fact2 #Shorts #viral #facts", "keywords": ["kw1", "kw2", "kw3"] }}""" resp = requests.post( "https://api.groq.com/openai/v1/chat/completions", headers={"Authorization": f"Bearer {os.environ['GROQ_API_KEY']}", "Content-Type": "application/json"}, json={"model": "llama-3.3-70b-versatile", "messages": [{"role": "user", "content": prompt}], "max_tokens": 1000, "temperature": 0.8}, timeout=30 ) resp.raise_for_status() raw = resp.json()["choices"][0]["message"]["content"].strip() raw = re.sub(r"^```[a-z]*\n?", "", raw) raw = re.sub(r"\n?```$", "", raw) data = json.loads(raw) data["source_topic"] = topic return data # ── VOICEOVER ───────────────────────────────────────────────────────────────── async def _tts(text: str, path: str): import edge_tts comm = edge_tts.Communicate(text, "en-US-GuyNeural") await comm.save(path) def generate_voice(script: str, filename: str) -> str: """Generate MP3 using edge-tts, upload to Supabase Storage.""" os.makedirs("/tmp/audio", exist_ok=True) local_path = f"/tmp/audio/{filename}.mp3" # Clean script script = re.sub(r'\[.*?\]|\(.*?\)', '', script) script = re.sub(r'\s+', ' ', script).strip() asyncio.run(_tts(script, local_path)) # Upload to Supabase Storage sb = get_sb() with open(local_path, "rb") as f: audio_bytes = f.read() storage_path = f"audio/{filename}.mp3" try: sb.storage.from_("videos").upload( storage_path, audio_bytes, {"content-type": "audio/mpeg", "upsert": "true"} ) public_url = sb.storage.from_("videos").get_public_url(storage_path) return public_url except Exception as e: print(f"Storage upload error: {e}") return None # ── MAIN PIPELINE ───────────────────────────────────────────────────────────── def run_pipeline(count: int = 3) -> list[dict]: """Run full pipeline: trends → scripts → voices → save to Supabase.""" print(f"[Pipeline] Starting — {count} videos") results = [] # Step 1: Trends print("[Pipeline] Fetching trends...") trends = get_trends(count * 2) # Step 2+3: Script + Voice for each for i, trend in enumerate(trends[:count]): print(f"[Pipeline] Processing {i+1}/{count}: {trend['topic'][:50]}") try: # Generate script script_data = generate_script(trend["topic"]) title = script_data.get("title", trend["topic"][:80]) print(f"[Pipeline] Script: '{title}'") # Generate voice slug = re.sub(r'[^\w]', '_', title)[:30] filename = f"{slug}_{i}_{int(datetime.now().timestamp())}" audio_url = generate_voice(script_data.get("script", ""), filename) print(f"[Pipeline] Voice: {audio_url}") # Save to Supabase sb = get_sb() row = { "title": title, "script": script_data.get("script", ""), "description": script_data.get("description", ""), "hashtags": script_data.get("hashtags", "#Shorts #viral"), "hook": script_data.get("hook", ""), "audio_url": audio_url, "status": "ready" } sb.table("videos").insert(row).execute() results.append(row) print(f"[Pipeline] ✅ Saved: '{title}'") except Exception as e: print(f"[Pipeline] ❌ Error on topic {i+1}: {e}") continue print(f"[Pipeline] Done! {len(results)}/{count} videos created.") return results
